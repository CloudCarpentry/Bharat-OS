# Syscall and Userspace capability matrix

set(BHARAT_ARCH_HAS_HW_SYSCALL_ENTRY 0)
set(BHARAT_ARCH_HAS_SYSCALL_EXTRACTION 0)
set(BHARAT_ARCH_HAS_SAFE_USERCOPY 0)
set(BHARAT_ARCH_HAS_COMPLETE_USERSPACE 0)
set(BHARAT_ARCH_HAS_SDK_SYSCALL_BACKEND 0)

if(BHARAT_ARCH_FAMILY MATCHES "X86" AND BHARAT_ARCH_BITS EQUAL 64)
    set(BHARAT_ARCH_HAS_HW_SYSCALL_ENTRY 1)
    set(BHARAT_ARCH_HAS_SYSCALL_EXTRACTION 1)
    set(BHARAT_ARCH_HAS_SAFE_USERCOPY 1)
    set(BHARAT_ARCH_HAS_COMPLETE_USERSPACE 1)
    set(BHARAT_ARCH_HAS_SDK_SYSCALL_BACKEND 1)
elseif(BHARAT_ARCH_FAMILY MATCHES "ARM" AND BHARAT_ARCH_BITS EQUAL 64)
    set(BHARAT_ARCH_HAS_HW_SYSCALL_ENTRY 1)
    set(BHARAT_ARCH_HAS_SYSCALL_EXTRACTION 1)
    set(BHARAT_ARCH_HAS_SAFE_USERCOPY 1)
    set(BHARAT_ARCH_HAS_COMPLETE_USERSPACE 1)
    set(BHARAT_ARCH_HAS_SDK_SYSCALL_BACKEND 1)
elseif(BHARAT_ARCH_FAMILY MATCHES "RISCV" AND BHARAT_ARCH_BITS EQUAL 64)
    set(BHARAT_ARCH_HAS_HW_SYSCALL_ENTRY 1)
    set(BHARAT_ARCH_HAS_SYSCALL_EXTRACTION 1)
    set(BHARAT_ARCH_HAS_SAFE_USERCOPY 1)
    set(BHARAT_ARCH_HAS_COMPLETE_USERSPACE 1)
    set(BHARAT_ARCH_HAS_SDK_SYSCALL_BACKEND 1)
elseif(BHARAT_ARCH_FAMILY MATCHES "ARM" AND BHARAT_ARCH_BITS EQUAL 32)
    set(BHARAT_ARCH_HAS_HW_SYSCALL_ENTRY 1)
    set(BHARAT_ARCH_HAS_SYSCALL_EXTRACTION 1)
    set(BHARAT_ARCH_HAS_SAFE_USERCOPY 1)
    set(BHARAT_ARCH_HAS_COMPLETE_USERSPACE 1)
    set(BHARAT_ARCH_HAS_SDK_SYSCALL_BACKEND 1)
elseif(BHARAT_ARCH_FAMILY MATCHES "RISCV" AND BHARAT_ARCH_BITS EQUAL 32)
    set(BHARAT_ARCH_HAS_HW_SYSCALL_ENTRY 1)
    set(BHARAT_ARCH_HAS_SYSCALL_EXTRACTION 1)
    set(BHARAT_ARCH_HAS_SAFE_USERCOPY 1)
    set(BHARAT_ARCH_HAS_COMPLETE_USERSPACE 1)
    set(BHARAT_ARCH_HAS_SDK_SYSCALL_BACKEND 1)
endif()

# Map back to legacy names to keep other build components working
set(BHARAT_ARCH_HAS_USER_SYSCALL_ENTRY ${BHARAT_ARCH_HAS_HW_SYSCALL_ENTRY})
set(BHARAT_ARCH_HAS_SYSCALL_EXTRACT ${BHARAT_ARCH_HAS_SYSCALL_EXTRACTION})
set(BHARAT_USERSPACE_SUPPORTED ${BHARAT_ARCH_HAS_COMPLETE_USERSPACE})

# Export variables to parent scope
set(BHARAT_ARCH_HAS_HW_SYSCALL_ENTRY ${BHARAT_ARCH_HAS_HW_SYSCALL_ENTRY} PARENT_SCOPE)
set(BHARAT_ARCH_HAS_SYSCALL_EXTRACTION ${BHARAT_ARCH_HAS_SYSCALL_EXTRACTION} PARENT_SCOPE)
set(BHARAT_ARCH_HAS_SAFE_USERCOPY ${BHARAT_ARCH_HAS_SAFE_USERCOPY} PARENT_SCOPE)
set(BHARAT_ARCH_HAS_COMPLETE_USERSPACE ${BHARAT_ARCH_HAS_COMPLETE_USERSPACE} PARENT_SCOPE)

set(BHARAT_ARCH_HAS_USER_SYSCALL_ENTRY ${BHARAT_ARCH_HAS_USER_SYSCALL_ENTRY} PARENT_SCOPE)
set(BHARAT_ARCH_HAS_SYSCALL_EXTRACT ${BHARAT_ARCH_HAS_SYSCALL_EXTRACT} PARENT_SCOPE)
set(BHARAT_ARCH_HAS_SDK_SYSCALL_BACKEND ${BHARAT_ARCH_HAS_SDK_SYSCALL_BACKEND} PARENT_SCOPE)
set(BHARAT_USERSPACE_SUPPORTED ${BHARAT_USERSPACE_SUPPORTED} PARENT_SCOPE)

# Add definitions for C code
add_compile_definitions(BHARAT_ARCH_HAS_HW_SYSCALL_ENTRY=${BHARAT_ARCH_HAS_HW_SYSCALL_ENTRY})
add_compile_definitions(BHARAT_ARCH_HAS_SYSCALL_EXTRACTION=${BHARAT_ARCH_HAS_SYSCALL_EXTRACTION})
add_compile_definitions(BHARAT_ARCH_HAS_SAFE_USERCOPY=${BHARAT_ARCH_HAS_SAFE_USERCOPY})
add_compile_definitions(BHARAT_ARCH_HAS_COMPLETE_USERSPACE=${BHARAT_ARCH_HAS_COMPLETE_USERSPACE})

add_compile_definitions(BHARAT_ARCH_HAS_USER_SYSCALL_ENTRY=${BHARAT_ARCH_HAS_USER_SYSCALL_ENTRY})
add_compile_definitions(BHARAT_ARCH_HAS_SYSCALL_EXTRACT=${BHARAT_ARCH_HAS_SYSCALL_EXTRACT})

# Validation: fail if userspace is requested but not supported for this arch
option(BHARAT_ENABLE_USERSPACE "Enable userspace support" ON)

if(BHARAT_ENABLE_USERSPACE AND NOT BHARAT_USERSPACE_SUPPORTED)
    message(FATAL_ERROR "Userspace/Syscall support requested (BHARAT_ENABLE_USERSPACE=ON) but not supported for ${BHARAT_ARCH_FAMILY}-${BHARAT_ARCH_BITS}")
endif()

# Build-time capability assertions
if(BHARAT_ARCH_HAS_COMPLETE_USERSPACE)
    if(NOT BHARAT_ARCH_HAS_HW_SYSCALL_ENTRY OR NOT BHARAT_ARCH_HAS_SYSCALL_EXTRACTION OR NOT BHARAT_ARCH_HAS_SAFE_USERCOPY)
        message(FATAL_ERROR "Architecture configuration error: BHARAT_ARCH_HAS_COMPLETE_USERSPACE set but missing required dependencies (HW entry, Extraction, or Safe Usercopy)")
    endif()
endif()
